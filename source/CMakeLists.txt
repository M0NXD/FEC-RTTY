cmake_minimum_required(VERSION 3.20)
project(fectty VERSION 0.46.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()
option(FECTTY_WITH_PORTAUDIO "Build PortAudio live audio backend" OFF)
option(FECTTY_WITH_HAMLIB "Build Hamlib direct/NET backend" OFF)
option(FECTTY_WITH_QT "Build Qt desktop GUI" OFF)
add_library(fectty_core
 src/crc16.cpp src/frame.cpp src/convolutional.cpp src/viterbi.cpp src/fsk4.cpp
 src/interleaver.cpp src/acquisition.cpp src/sync.cpp src/timing.cpp src/stream_receiver.cpp src/net_rigctl.cpp
 src/session.cpp src/radio_controller.cpp src/frame_receiver.cpp src/spectrum.cpp src/settings.cpp src/channel.cpp src/macros.cpp src/qso_log.cpp
 src/reliable_receiver.cpp src/arq.cpp src/clock_drift.cpp src/link_metrics.cpp src/diagnostics.cpp
 src/wav.cpp src/level_meter.cpp src/timing_tracker.cpp src/interop.cpp src/waterfall_spectrum.cpp)
target_include_directories(fectty_core PUBLIC include)
if(WIN32)
 target_sources(fectty_core PRIVATE src/omnirig.cpp)
 target_link_libraries(fectty_core PUBLIC ws2_32 ole32 oleaut32)
endif()
if(FECTTY_WITH_HAMLIB)
 set(HAMLIB_ROOT "" CACHE PATH "Hamlib development SDK root")
 if(HAMLIB_ROOT)
  find_path(HAMLIB_INCLUDE_DIR hamlib/rig.h PATHS "${HAMLIB_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(HAMLIB_LIBRARY NAMES hamlib libhamlib PATHS "${HAMLIB_ROOT}/lib/gcc" "${HAMLIB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(fectty_hamlib_sdk INTERFACE)
  target_include_directories(fectty_hamlib_sdk INTERFACE "${HAMLIB_INCLUDE_DIR}")
  target_link_libraries(fectty_hamlib_sdk INTERFACE "${HAMLIB_LIBRARY}")
 else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(HAMLIB REQUIRED IMPORTED_TARGET hamlib)
  add_library(fectty_hamlib_sdk INTERFACE)
  target_link_libraries(fectty_hamlib_sdk INTERFACE PkgConfig::HAMLIB)
 endif()
 target_sources(fectty_core PRIVATE src/hamlib_rig.cpp)
 target_link_libraries(fectty_core PUBLIC fectty_hamlib_sdk)
 target_compile_definitions(fectty_core PUBLIC FECTTY_HAS_HAMLIB=1)
endif()
if(FECTTY_WITH_PORTAUDIO)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
 target_sources(fectty_core PRIVATE src/portaudio_io.cpp)
 target_link_libraries(fectty_core PUBLIC PkgConfig::PORTAUDIO)
 target_compile_definitions(fectty_core PUBLIC FECTTY_HAS_PORTAUDIO=1)
 add_executable(fectty-portaudio-smoke src/portaudio_smoke.cpp)
 target_link_libraries(fectty-portaudio-smoke PRIVATE fectty_core)
endif()
if(MSVC)
 target_compile_options(fectty_core PRIVATE /W4)
else()
 target_compile_options(fectty_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(fectty-sim src/main.cpp)
target_link_libraries(fectty-sim PRIVATE fectty_core)
target_compile_definitions(fectty-sim PRIVATE FECTTY_PROJECT_VERSION="${PROJECT_VERSION}")
if(FECTTY_WITH_QT)
 find_package(Qt6 REQUIRED COMPONENTS Widgets)
 add_executable(fectty-gui src/gui_main.cpp src/winmm_io.cpp src/waterfall_widget.cpp)
 target_link_libraries(fectty-gui PRIVATE fectty_core Qt6::Widgets)
 if(WIN32)
  target_link_libraries(fectty-gui PRIVATE winmm)
 endif()
 target_compile_definitions(fectty-gui PRIVATE FECTTY_HAS_QT=1 FECTTY_PROJECT_VERSION="${PROJECT_VERSION}")
endif()
add_executable(fectty-benchmark src/benchmark.cpp)
target_link_libraries(fectty-benchmark PRIVATE fectty_core)
add_executable(fectty-tests tests/tests.cpp)
target_link_libraries(fectty-tests PRIVATE fectty_core)
add_executable(fectty-audit-tests tests/audit_tests.cpp)
target_link_libraries(fectty-audit-tests PRIVATE fectty_core)
add_executable(fectty-cat-tests tests/cat_tests.cpp)
target_link_libraries(fectty-cat-tests PRIVATE fectty_core)
if(WIN32)
 add_executable(fectty-live src/live.cpp)
 target_link_libraries(fectty-live PRIVATE fectty_core winmm)
 add_executable(fectty-audio-smoke src/audio_smoke.cpp src/winmm_io.cpp)
 target_link_libraries(fectty-audio-smoke PRIVATE fectty_core winmm)
 add_executable(fectty-noise-prefix-bench tests/noise_prefix_bench.cpp src/winmm_io.cpp)
 target_link_libraries(fectty-noise-prefix-bench PRIVATE fectty_core winmm)
endif()
enable_testing()
add_test(NAME fectty-tests COMMAND fectty-tests)
add_test(NAME fectty-audit-tests COMMAND fectty-audit-tests)
add_test(NAME fectty-cat-tests COMMAND fectty-cat-tests)
set_tests_properties(fectty-cat-tests PROPERTIES TIMEOUT 60)
set_tests_properties(fectty-tests fectty-audit-tests PROPERTIES TIMEOUT 120)
if(FECTTY_WITH_QT)
 add_executable(fectty-gui-text-tests tests/gui_text_tests.cpp)
 target_include_directories(fectty-gui-text-tests PRIVATE include)
 target_link_libraries(fectty-gui-text-tests PRIVATE Qt6::Widgets)
 add_test(NAME fectty-gui-text-tests COMMAND fectty-gui-text-tests)
 add_executable(fectty-waterfall-tests tests/waterfall_tests.cpp src/waterfall_widget.cpp)
 target_link_libraries(fectty-waterfall-tests PRIVATE fectty_core Qt6::Widgets)
 add_test(NAME fectty-waterfall-tests COMMAND fectty-waterfall-tests)
 set_tests_properties(fectty-gui-text-tests PROPERTIES TIMEOUT 30)
 set_tests_properties(fectty-waterfall-tests PROPERTIES TIMEOUT 30)
 if(NOT WIN32)
  set_tests_properties(fectty-gui-text-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(fectty-waterfall-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 endif()
endif()
install(TARGETS fectty-sim fectty-benchmark RUNTIME DESTINATION bin)
if(WIN32)
 install(TARGETS fectty-live RUNTIME DESTINATION bin)
endif()
if(FECTTY_WITH_PORTAUDIO)
 install(TARGETS fectty-portaudio-smoke RUNTIME DESTINATION bin)
endif()
if(FECTTY_WITH_QT)
 install(TARGETS fectty-gui RUNTIME DESTINATION bin)
endif()
install(DIRECTORY include/fectty DESTINATION include)
